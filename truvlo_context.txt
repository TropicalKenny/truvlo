/**
 * Safely stringify any value for error messages
 */
function safeStringify(obj: any): string {
  if (obj === null) return 'null';
  if (obj === undefined) return 'undefined';
  if (typeof obj === 'string') return obj;
  if (typeof obj === 'number' || typeof obj === 'boolean') return String(obj);
  
  try {
    // Handle Error objects specially
    if (obj instanceof Error) {
      return obj.message || obj.toString() || 'Error';
    }
    
    // Try to get message property
    if (obj.message && typeof obj.message === 'string') {
      return obj.message;
    }
    
    // Try JSON stringify with a replacer for circular refs
    const seen = new WeakSet();
    const str = JSON.stringify(obj, (key, value) => {
      if (typeof value === 'object' && value !== null) {
        if (seen.has(value)) return '[Circular]';
        seen.add(value);
      }
      return value;
    });
    
    if (str && str !== '{}' && str !== '[]') {
      return str.length > 300 ? str.substring(0, 300) + '...' : str;
    }
  } catch {
    // Ignore stringify errors
  }
  
  // Last resort
  try {
    return String(obj);
  } catch {
    return 'Unknown error';
  }
}

/**
 * EscrowPay SDK for invoice and escrow management
 */
export class EscrowPaySDK {
  private readonly provider: Provider;
  private program: Program<Idl> | null = null;
  private readonly usdcMint: PublicKey;
  private readonly programId: PublicKey;
  private coder: BorshCoder | null = null;
  private initialized: boolean = false;
  private initError: string | null = null;

  constructor(provider: Provider, useMainnet: boolean = false) {
    this.provider = provider;
    this.programId = PROGRAM_ID;
    this.usdcMint = useMainnet ? USDC_MINT_MAINNET : USDC_MINT_DEVNET;
    
    // Initialize program with explicit program ID for Anchor 0.30+ compatibility
    try {
      this.program = new Program(IDL as Idl, this.provider);
      this.coder = new BorshCoder(IDL as Idl);
      this.initialized = true;
    } catch (e) {
      console.error('Failed to initialize Anchor program:', e);
      this.initError = e instanceof Error ? e.message : 'Failed to initialize program';
      this.initialized = false;
      // Don't throw - let the caller handle it gracefully
    }
  }

  /**
   * Check if SDK is properly initialized
   */
  isInitialized(): boolean {
    return this.initialized;
  }

  /**
   * Get initialization error if any
   */
  getInitError(): string | null {
    return this.initError;
  }

  /**
   * Decode account data manually using BorshCoder
   */
  private decodeInvoiceAccount(data: Buffer): any {
    if (!this.coder) return null;
    try {
      // Try Invoice (capital I) first - matches IDL type name
      return this.coder.accounts.decode('Invoice', data);
    } catch (e) {
      // Fallback to lowercase
      try {
        return this.coder.accounts.decode('invoice', data);
      } catch {
        console.error('Failed to decode invoice account:', e);
        return null;
      }
    }
  }

  /**
   * Parse raw invoice account data into InvoiceData
   */
  private parseInvoiceData(raw: any): InvoiceData | null {
    if (!raw) return null;
    
    try {
      // Parse token type
      let tokenType = TokenType.USDC;
      if (raw.tokenType?.sol !== undefined || raw.token_type?.sol !== undefined) {
        tokenType = TokenType.SOL;
      }

      // Parse status
      let status = InvoiceStatus.Unpaid;
      const rawStatus = raw.status || {};
      if (rawStatus.partiallyPaid !== undefined || rawStatus.partially_paid !== undefined) {
        status = InvoiceStatus.PartiallyPaid;
      } else if (rawStatus.fullyPaid !== undefined || rawStatus.fully_paid !== undefined) {
        status = InvoiceStatus.FullyPaid;
      } else if (rawStatus.released !== undefined) {
        status = InvoiceStatus.Released;
      }

      return {
        bump: raw.bump,
        escrowBump: raw.escrowBump ?? raw.escrow_bump ?? 0,
        invoiceId: raw.invoiceId ?? raw.invoice_id ?? '',
        creator: raw.creator,
        client: raw.client,
        amount: raw.amount,
        amountPaid: raw.amountPaid ?? raw.amount_paid ?? new BN(0),
        amountReleased: raw.amountReleased ?? raw.amount_released ?? new BN(0),
        tokenType,
        description: raw.description ?? '',
        clientEmail: raw.clientEmail ?? raw.client_email ?? '',
        status,
        createdAt: raw.createdAt ?? raw.created_at ?? new BN(0),
        paidAt: raw.paidAt ?? raw.paid_at ?? null,
        releasedAt: raw.releasedAt ?? raw.released_at ?? null,
        releaseSchedule: raw.releaseSchedule ?? raw.release_schedule ?? [],
        currentReleaseIndex: raw.currentReleaseIndex ?? raw.current_release_index ?? 0,
        completionRequestedAt: raw.completionRequestedAt ?? raw.completion_requested_at ?? null,
        clientApproved: raw.clientApproved ?? raw.client_approved ?? false,
        disputeRaisedAt: raw.disputeRaisedAt ?? raw.dispute_raised_at ?? null,
      };
    } catch (e) {
      console.error('Failed to parse invoice data:', e);
      return null;
    }
  }

  /**
   * Calculate payment breakdown with platform fee
   */
  calculatePaymentBreakdown(paymentAmount: number, tokenType: TokenType, solPriceUsd?: number): PaymentBreakdown {
    let platformFee: number;
    let feePercentage: number;
    
    if (tokenType === TokenType.USDC) {
      // USDC: exact dollar comparison
      if (paymentAmount < PLATFORM_FEE_THRESHOLD_USD) {
        platformFee = PLATFORM_FLAT_FEE_USD;
        feePercentage = (PLATFORM_FLAT_FEE_USD / paymentAmount) * 100;
      } else {
        platformFee = paymentAmount * (PLATFORM_FEE_BPS / 10000);
        feePercentage = PLATFORM_FEE_BPS / 100;
      }
    } else {
      // SOL: use live price if available, fallback to 1%
      if (solPriceUsd && solPriceUsd > 0) {
        const valueUsd = paymentAmount * solPriceUsd;
        if (valueUsd < PLATFORM_FEE_THRESHOLD_USD) {
          platformFee = PLATFORM_FLAT_FEE_USD / solPriceUsd; //  in SOL
          feePercentage = (platformFee / paymentAmount) * 100;
        } else {
          platformFee = paymentAmount * (PLATFORM_FEE_BPS / 10000);
          feePercentage = PLATFORM_FEE_BPS / 100;
        }
      } else {
        // Fallback: always 1% if no price available
        platformFee = paymentAmount * (PLATFORM_FEE_BPS / 10000);
        feePercentage = PLATFORM_FEE_BPS / 100;
      }
    }
    
    const totalPayable = paymentAmount + platformFee;
    return {
      invoiceAmount: paymentAmount,
      platformFee: parseFloat(platformFee.toFixed(tokenType === TokenType.USDC ? 6 : 9)),
      totalPayable: parseFloat(totalPayable.toFixed(tokenType === TokenType.USDC ? 6 : 9)),
      feePercentage,
    };
  }

  /**
   * Convert to smallest unit based on token type
   */
  private toSmallestUnit(amount: number, tokenType: TokenType): BN {
    const decimals = tokenType === TokenType.USDC ? 6 : 9;
    // Use string-based conversion to avoid floating point precision loss
    // e.g. 1 SOL: Math.floor(1 * 10^9) = 999999999 due to float â€” this fixes it
    const amountStr = amount.toFixed(decimals);
    const [intPart, fracPart = ''] = amountStr.split('.');
    const paddedFrac = fracPart.padEnd(decimals, '0').slice(0, decimals);
    const smallestUnit = BigInt(intPart) * BigInt(10 ** decimals) + BigInt(paddedFrac);
    return new BN(smallestUnit.toString());
  }

  /**
   * Convert from smallest unit based on token type
   */
  private fromSmallestUnit(amount: BN, tokenType: TokenType): number {
    const decimals = tokenType === TokenType.USDC ? 6 : 9;
    try {
      return amount.toNumber() / Math.pow(10, decimals);
    } catch {
      return parseInt(amount.toString()) / Math.pow(10, decimals);
    }
  }

  /**
   * Test RPC connection
   */
  private async testConnection(): Promise<boolean> {
    try {
      if (!this.provider?.connection) return false;
      const { value } = await this.provider.connection.getLatestBlockhashAndContext('finalized');
      return !!(value && value.blockhash);
    } catch {
      return false;
    }
  }

  /**
   * Parse errors to human-readable string messages
   * ALWAYS returns a string, never an object
   */
  private parseError(error: any): string {
    // Handle null/undefined
    if (!error) {
      return 'Unknown error occurred.';
    }
    
    // Already a string
    if (typeof error === 'string') {
      return error;
    }
    
    // Get message safely
    let message = '';
    try {
      message = error.message || error.msg || '';
      if (typeof message !== 'string') {
        message = safeStringify(message);
      }
    } catch {
      message = '';
    }
    
    // Get string representation safely
    let errorStr = '';
    try {
      errorStr = typeof error.toString === 'function' ? error.toString() : '';
      if (errorStr === '[object Object]') {
        errorStr = '';
      }
    } catch {
      errorStr = '';
    }
    
    // Check for user rejection
    if (message.includes('User rejected') || message.includes('rejected the request') ||
        errorStr.includes('User rejected') || errorStr.includes('rejected the request')) {
      return 'Transaction was cancelled.';
    }
    
    // Check for insufficient funds / no prior credit (wallet has no SOL)
    if (message.toLowerCase().includes('insufficient') || 
        errorStr.toLowerCase().includes('insufficient') ||
        message.includes('no record of a prior credit') ||
        errorStr.includes('no record of a prior credit') ||
        message.includes('Attempt to debit an account') ||
        errorStr.includes('Attempt to debit an account')) {
      return 'Insufficient SOL balance. Please add SOL to your wallet for transaction fees.';
    }
    
    // Check for simulation failed
    if (message.includes('Simulation failed') || errorStr.includes('Simulation failed')) {
      // Extract the actual error from the message
      if (message.includes('no record of a prior credit') || message.includes('Attempt to debit')) {
        return 'Insufficient SOL balance. Please add SOL to your wallet for transaction fees.';
      }
      if (message.includes('custom program error')) {
        const match = message.match(/custom program error: (0x[0-9a-fA-F]+|\d+)/);
        if (match) return `Program error: ${match[1]}`;
      }
      return 'Transaction simulation failed. Please try again.';
    }
    
    // Anchor program errors
    try {
      if (error.error?.errorCode?.code) {
        const msg = error.error.errorMessage || error.error.errorCode.name;
        if (msg && typeof msg === 'string') {
          return msg;
        }
        return 'Program error: ' + String(error.error.errorCode.code);
      }
    } catch {
      // Ignore
    }
    
    // Check for program logs
    try {
      if (error.logs && Array.isArray(error.logs)) {
        for (const log of error.logs) {
          if (typeof log === 'string') {
            if (log.includes('Error Code:')) {
              const match = log.match(/Error Code: (\w+)/);
              if (match) return 'Error: ' + match[1];
            }
            if (log.includes('Error Message:')) {
              const match = log.match(/Error Message: (.+)/);
              if (match) return match[1];
            }
          }
        }
      }
    } catch {
      // Ignore
    }
    
    // Simulation errors
    try {
      if (error.simulationResponse?.err) {
        const simErr = error.simulationResponse.err;
        if (typeof simErr === 'string') {
          return simErr;
        }
        if (simErr?.InstructionError) {
          const instErr = simErr.InstructionError[1];
          if (typeof instErr === 'string') {
            return instErr;
          }
          if (instErr?.Custom !== undefined) {
            return 'Program error code: ' + String(instErr.Custom);
          }
        }
        return 'Transaction simulation failed.';
      }
    } catch {
      // Ignore
    }
    
    // Return message if we have one
    if (message && message.length > 0) {
      // Clean up AnchorError messages
      if (message.includes('AnchorError')) {
        const match = message.match(/Error Message: ([^.]+)/);
        if (match) return match[1];
      }
      return message.length > 200 ? message.substring(0, 200) + '...' : message;
    }
    
    // Try to extract something useful from error string
    if (errorStr && errorStr.length > 0 && errorStr !== '[object Object]') {
      return errorStr.length > 200 ? errorStr.substring(0, 200) + '...' : errorStr;
    }
    
    // Last resort
    return 'Transaction failed. Please try again.';
  }

  /**
   * Log error for debugging (always outputs strings)
   */
  private logError(context: string, error: any): void {
    const safeError = {
      context,
      name: String(error?.name || 'Unknown'),
      message: String(error?.message || 'No message'),
      parsed: this.parseError(error),
    };
    console.error('SDK Error:', JSON.stringify(safeError, null, 2));
  }

  /**
   * Generate unique invoice ID
   */
  generateInvoiceId(): string {
    const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
    let result = 'INV-';
    for (let i = 0; i < 8; i++) {
      result += chars.charAt(Math.floor(Math.random() * chars.length));
    }
    return result;
  }

  /**
   * Create a new invoice (USDC or SOL)
   */
  async createInvoice(params: CreateInvoiceParams): Promise<SDKResult<{ 
    signature: string; 
    invoiceAddress: string;
    invoiceId: string;
  }>> {
    if (!this.provider.publicKey) {
      return { success: false, error: "Wallet not connected" };
    }

    if (!this.initialized || !this.program) {
      return { success: false, error: this.initError || "SDK not initialized" };
    }

    try {
      if (!(await this.testConnection())) {
        return { success: false, error: "Network unavailable" };
      }

      // Validate params
      if (!params.invoiceId?.trim() || params.invoiceId.length > 32) {
        return { success: false, error: "Invalid invoice ID" };
      }
      if (params.amount <= 0) {
        return { success: false, error: "Amount must be greater than 0" };
      }
      if (!params.description?.trim() || params.description.length > 256) {
        return { success: false, error: "Invalid description" };
      }
      if (!params.clientEmail?.trim() || params.clientEmail.length > 128) {
        return { success: false, error: "Invalid client email" };
      }

      const [invoiceAddress] = await this.getInvoicePDA(params.invoiceId, this.provider.publicKey);
      const amountSmallestUnit = this.toSmallestUnit(params.amount, params.tokenType);

      let tx: string;

      if (params.tokenType === TokenType.SOL) {
        tx = await this.program.methods
          .initializeConfigSol(
            params.invoiceId,
            amountSmallestUnit,
            params.description.trim(),
            params.clientEmail.trim()
          )
          .accounts({
            invoice: invoiceAddress,
            creator: this.provider.publicKey,
            systemProgram: SystemProgram.programId,
          })
          .rpc();
      } else {
        const [escrowAddress] = await this.getEscrowPDA(params.invoiceId, this.provider.publicKey);
        
        tx = await this.program.methods
          .initializeConfig(
            params.invoiceId,
            amountSmallestUnit,
            params.description.trim(),
            params.clientEmail.trim()
          )
          .accounts({
            invoice: invoiceAddress,
            escrow: escrowAddress,
            usdcMint: this.usdcMint,
            creator: this.provider.publicKey,
            tokenProgram: TOKEN_PROGRAM_ID,
            systemProgram: SystemProgram.programId,
            rent: RENT_SYSVAR_ID,
          })
          .rpc();
      }

      return {
        success: true,
        data: {
          signature: tx,
          invoiceAddress: invoiceAddress.toString(),
          invoiceId: params.invoiceId,
        }
      };
    } catch (error: any) {
      this.logError('createInvoice', error);
      const errorMessage = this.parseError(error);
      return { success: false, error: errorMessage };
    }
  }

  /**
   * Pay an invoice (supports partial/milestone payments)
   */
  async payInvoice(
    invoiceId: string, 
    creatorPubkey: PublicKey, 
    paymentAmount: number,
    tokenType: TokenType,
    solPriceUsd?: number
  ): Promise<SDKResult<{ signature: string; breakdown: PaymentBreakdown }>> {
    if (!this.provider.publicKey) {
      return { success: false, error: "Wallet not connected" };
    }

    if (!this.initialized || !this.program) {
      return { success: false, error: this.initError || "SDK not initialized" };
    }

    try {
      if (!(await this.testConnection())) {
        return { success: false, error: "Network unavailable" };
      }

      const [invoiceAddress] = await this.getInvoicePDA(invoiceId, creatorPubkey);
      const paymentSmallestUnit = this.toSmallestUnit(paymentAmount, tokenType);
      // Fetch SOL price if not provided and token is SOL
      const resolvedSolPrice = tokenType === TokenType.SOL && !solPriceUsd
        ? await this.getSolPriceUsd()
        : solPriceUsd ?? null;
      const breakdown = this.calculatePaymentBreakdown(paymentAmount, tokenType, resolvedSolPrice ?? undefined);

      let tx: string;

      if (tokenType === TokenType.SOL) {
        const feeSmallestUnit = this.toSmallestUnit(breakdown.platformFee, TokenType.SOL);
        tx = await this.program.methods
          .payInvoiceSol(invoiceId, paymentSmallestUnit, feeSmallestUnit)
          .accounts({
            invoice: invoiceAddress,
            platformTreasury: PLATFORM_TREASURY,
            client: this.provider.publicKey,
            systemProgram: SystemProgram.programId,
          })
          .rpc();
      } else {
        const [escrowAddress] = await this.getEscrowPDA(invoiceId, creatorPubkey);
        const clientToken = getAssociatedTokenAddressSync(this.usdcMint, this.provider.publicKey);
        const treasuryToken = getAssociatedTokenAddressSync(this.usdcMint, PLATFORM_TREASURY);

        tx = await this.program.methods
          .payInvoice(invoiceId, paymentSmallestUnit)
          .accounts({
            invoice: invoiceAddress,
            escrow: escrowAddress,
            clientToken: clientToken,
            platformTreasuryToken: treasuryToken,
            usdcMint: this.usdcMint,
            client: this.provider.publicKey,
            tokenProgram: TOKEN_PROGRAM_ID,
          })
          .rpc();
      }

      return { success: true, data: { signature: tx, breakdown } };
    } catch (error: any) {
      this.logError('payInvoice', error);
      const errorMessage = this.parseError(error);
      return { success: false, error: errorMessage };
    }
  }

  /**
   * Set release schedule â€” called by client after paying full amount
   * scheduleBps: array of basis points per installment (must sum to 10000)
   */
  async setReleaseSchedule(
    invoiceId: string,
    creatorPubkey: PublicKey,
    scheduleBps: number[]
  ): Promise<SDKResult<{ signature: string }>> {
    if (!this.provider.publicKey) return { success: false, error: 'Wallet not connected' };
    if (!this.initialized || !this.program) return { success: false, error: this.initError || 'SDK not initialized' };
    try {
      if (!(await this.testConnection())) return { success: false, error: 'Network unavailable' };
      const [invoiceAddress] = await this.getInvoicePDA(invoiceId, creatorPubkey);
      const tx = await this.program.methods
        .setReleaseSchedule(invoiceId, scheduleBps)
        .accounts({ invoice: invoiceAddress, client: this.provider.publicKey })

  /**

  async fetchInvoicesByCreator(creator?: PublicKey): Promise<SDKResult<Array<{ publicKey: PublicKey; account: InvoiceData }>>> {
    const targetCreator = creator || this.provider.publicKey;
    if (!targetCreator) {
      return { success: false, error: "No creator provided" };
    }

    try {
      if (!this.initialized || !this.program) {

        if (invoiceAccessor) {
          const allInvoices = await invoiceAccessor.all();
          if (allInvoices?.length) {
            const filtered = allInvoices.filter((inv: any) => {
              try {
                return inv?.account?.creator?.toString() === targetCreator.toString();
              } catch {
                return false;
              }
            });

            const mapped = filtered.map((inv: any) => {
              const parsed = this.parseInvoiceData(inv.account);

                publicKey: inv.publicKey,

            }).filter((inv: any) => inv.account !== null);

            return { success: true, data: mapped };
          }
        }
      } catch (anchorError) {
        console.log('Anchor fetchAll failed, trying direct RPC:', anchorError);
      }

      // Fallback: Direct RPC call with getProgramAccounts
      const accounts = await this.provider.connection.getProgramAccounts(this.programId, {
        filters: [
          // Filter by creator (offset 10 for creator pubkey after discriminator + bumps)
          {
            memcmp: {
              offset: 10, // 8 bytes discriminator + 2 bytes for bumps
              bytes: targetCreator.toBase58(),
            },
          },
        ],
      });

      if (!accounts?.length) {
        return { success: true, data: [] };
      }

      const mapped = accounts.map((acc) => {
        try {
          const decoded = this.decodeInvoiceAccount(acc.account.data);
          const parsed = this.parseInvoiceData(decoded);
          if (parsed) {
            return {
              publicKey: acc.pubkey,
              account: parsed,
            };
          }
        } catch (e) {
          console.error('Failed to decode account:', e);
        }
        return null;
      }).filter((inv): inv is { publicKey: PublicKey; account: InvoiceData } => inv !== null);

      return { success: true, data: mapped };
    } catch (error: any) {
      this.logError('fetchInvoicesByCreator', error);
      return { success: true, data: [] }; // Return empty array instead of error for better UX
    }
  }

  /**
   * Fetch all invoices where the connected wallet is the CLIENT who paid.
   * Uses fetchAll + client-side filter because invoice_id is variable-length,
   * making memcmp byte offsets unreliable for the client field.
   */
  async fetchInvoicesByClient(client?: PublicKey): Promise<SDKResult<Array<{ publicKey: PublicKey; account: InvoiceData }>>> {
    const targetClient = client || this.provider.publicKey;
    if (!targetClient) return { success: false, error: 'No client provided' };
    if (!this.initialized || !this.program) return { success: false, error: this.initError || 'SDK not initialized' };
    try {
      if (!(await this.testConnection())) return { success: false, error: 'Network unavailable' };

      const targetStr = targetClient.toString();

      // Try Anchor accessor first â€” fetch all, filter client-side
      try {
        const invoiceAccessor = (this.program.account as any).invoice || (this.program.account as any).Invoice;
        if (invoiceAccessor) {
          const allInvoices = await invoiceAccessor.all();
          if (allInvoices?.length) {
            // Debug: log first invoice to inspect client field shape
            if (allInvoices.length > 0) {
              const sample = allInvoices[0]?.account;
              console.log('[fetchInvoicesByClient] sample client field:', JSON.stringify(sample?.client), 'target:', targetStr);
            }
            const filtered = allInvoices.filter((inv: any) => {
              try {
                const clientField = inv?.account?.client;
                if (!clientField) return false;
                // Anchor Option<Pubkey> can come as:
                // 1. PublicKey directly (most common)
                // 2. { key: PublicKey } (some versions)
                // 3. String already
                let clientKey: string;
                if (typeof clientField === 'string') {
                  clientKey = clientField;
                } else if (clientField.toBase58) {
                  clientKey = clientField.toBase58();
                } else if (clientField.toString) {
                  clientKey = clientField.toString();
                } else if (clientField.key) {
                  clientKey = clientField.key.toString();
                } else {
                  clientKey = String(clientField);
                }
                return clientKey === targetStr;
              } catch { return false; }
            });
            const mapped = filtered.map((inv: any) => {
              const parsed = this.parseInvoiceData(inv.account);
              return { publicKey: inv.publicKey, account: parsed as InvoiceData };
            }).filter((inv: any) => inv.account !== null);
            return { success: true, data: mapped };
          }
        }
      } catch (anchorError) {
        console.log('Anchor fetchAll failed for client invoices, trying RPC fallback:', anchorError);
      }

      // Fallback: fetch ALL program accounts, decode and filter client-side
      // We cannot use memcmp here because invoice_id is variable-length String,
      // meaning the client pubkey offset shifts per invoice.
      const accounts = await this.provider.connection.getProgramAccounts(this.programId);

      if (!accounts?.length) return { success: true, data: [] };

      const mapped = accounts.map((acc) => {
        try {
          const decoded = this.decodeInvoiceAccount(acc.account.data);
          const parsed = this.parseInvoiceData(decoded);
          if (!parsed) return null;
          // Filter: client field must match target wallet
          if (!parsed.client) return null;
          if (parsed.client.toString() !== targetStr) return null;
          return { publicKey: acc.pubkey, account: parsed };
        } catch { return null; }
      }).filter(Boolean) as Array<{ publicKey: PublicKey; account: InvoiceData }>;

      return { success: true, data: mapped };
    } catch (error: any) {
      this.logError('fetchInvoicesByClient', error);
      return { success: false, error: this.parseError(error) };
    }
  }

  /**
   * Returns true if the auto-release timeout has passed for a given invoice
   * (7 days after completionRequestedAt)
   */
  isAutoReleaseReady(invoice: import('./escrowPaySDK').InvoiceData): boolean {
    if (!invoice.completionRequestedAt) return false;
    const requestedAt = invoice.completionRequestedAt.toNumber();
    const now = Math.floor(Date.now() / 1000);
    return now - requestedAt > 7 * 24 * 60 * 60;
  }

  /**
   * Returns seconds remaining until auto-release (0 if already passed)
   */
  autoReleaseSecondsRemaining(invoice: import('./escrowPaySDK').InvoiceData): number {
    if (!invoice.completionRequestedAt) return -1;
    const requestedAt = invoice.completionRequestedAt.toNumber();
    const now = Math.floor(Date.now() / 1000);
    const elapsed = now - requestedAt;
    const timeout = 7 * 24 * 60 * 60;
    return Math.max(0, timeout - elapsed);
  }

    return this.program.programId;
  }

function formatCountdown(seconds: number): string {
  if (seconds <= 0) return 'Auto-releasing soon';
  const days = Math.floor(seconds / 86400);
  const hours = Math.floor((seconds % 86400) / 3600);
  const mins = Math.floor((seconds % 3600) / 60);
  if (days > 0) return `${days}d ${hours}h remaining`;
  if (hours > 0) return `${hours}h ${mins}m remaining`;
  return `${mins}m remaining`;
}

function formatCountdown(seconds: number): string {
  if (seconds <= 0) return 'Auto-releasing soon';
  const days = Math.floor(seconds / 86400);
  const hours = Math.floor((seconds % 86400) / 3600);
  const mins = Math.floor((seconds % 3600) / 60);
  if (days > 0) return `${days}d ${hours}h remaining`;
  if (hours > 0) return `${hours}h ${mins}m remaining`;
  return `${mins}m remaining`;
}

 * Safely convert any error to a string message
 */
function getErrorMessage(error: unknown): string {
  if (!error) return 'Unknown error';
  if (typeof error === 'string') return error;
  
  // Check for Error object
  if (error instanceof Error) {
    return error.message || 'An error occurred';
  }
  
  // Check for object with message
  if (typeof error === 'object') {
    const obj = error as Record<string, any>;
    if (typeof obj.message === 'string') return obj.message;
    if (typeof obj.msg === 'string') return obj.msg;
    if (typeof obj.error === 'string') return obj.error;
    
    // Try to stringify
    try {
      const str = JSON.stringify(error);
      if (str && str !== '{}') {
        return str.length > 150 ? str.substring(0, 150) + '...' : str;
      }
    } catch {
      // Ignore
    }
  }
  
  return 'An error occurred';
}

import React, { createContext, useContext, useState, useCallback } from 'react';
import { X, Bell, CheckCheck } from 'lucide-react';
import { Button } from '@/components/ui/button';

interface Notification {
  id: string;
  type: 'info' | 'success' | 'warning' | 'error';
  title: string;
  message?: string;
  timestamp: Date;
  read: boolean;
}

interface NotificationContextType {
  notifications: Notification[];
  unreadCount: number;
  addNotification: (notification: Omit<Notification, 'id' | 'timestamp' | 'read'>) => void;
  removeNotification: (id: string) => void;
  markAsRead: (id: string) => void;
  markAllAsRead: () => void;
  clearAll: () => void;
}

const NotificationContext = createContext<NotificationContextType | null>(null);

export const useNotifications = () => {
  const context = useContext(NotificationContext);
  if (!context) throw new Error('useNotifications must be used within NotificationProvider');
  return context;
};

interface NotificationProviderProps {
  children: React.ReactNode;
}

export const NotificationProvider: React.FC<NotificationProviderProps> = ({ children }) => {
  const [notifications, setNotifications] = useState<Notification[]>([]);

  const addNotification = useCallback((notification: Omit<Notification, 'id' | 'timestamp' | 'read'>) => {
    const newNotification: Notification = {
      ...notification,
      id: Math.random().toString(36).substr(2, 9),
      timestamp: new Date(),
      read: false,
    };
    // Keep last 100 notifications
    setNotifications(prev => [newNotification, ...prev].slice(0, 100));
  }, []);

  const removeNotification = useCallback((id: string) => {
    setNotifications(prev => prev.filter(n => n.id !== id));
  }, []);

  const markAsRead = useCallback((id: string) => {
    setNotifications(prev => prev.map(n => n.id === id ? { ...n, read: true } : n));
  }, []);

  const markAllAsRead = useCallback(() => {
    setNotifications(prev => prev.map(n => ({ ...n, read: true })));
  }, []);

  const clearAll = useCallback(() => {
    setNotifications([]);
  }, []);

  const unreadCount = notifications.filter(n => !n.read).length;

  return (
    <NotificationContext.Provider value={{
      notifications,
      unreadCount,
      addNotification,
      removeNotification,
      markAsRead,
      markAllAsRead,
      clearAll,
    }}>
      {children}
    </NotificationContext.Provider>
  );
};

    const opacity = read ? 'opacity-60' : '';

      case 'success': return `bg-green-50 dark:bg-green-900/20 ${opacity}`;
      case 'warning': return `bg-amber-50 dark:bg-amber-900/20 ${opacity}`;
      case 'error': return `bg-red-50 dark:bg-red-900/20 ${opacity}`;
      default: return `bg-blue-50 dark:bg-blue-900/20 ${opacity}`;

    switch (type) {
      case 'success': return 'bg-green-500';
      case 'warning': return 'bg-amber-500';
      case 'error': return 'bg-red-500';
      default: return 'bg-blue-500';
    }
  };

  const formatTime = (date: Date) => {
    const now = new Date();
    const diff = now.getTime() - date.getTime();
    const mins = Math.floor(diff / 60000);
    const hours = Math.floor(diff / 3600000);
    const days = Math.floor(diff / 86400000);
    if (mins < 1) return 'just now';
    if (mins < 60) return `${mins}m ago`;
    if (hours < 24) return `${hours}h ago`;
    return `${days}d ago`;
  };

import React from 'react';
import Header from '@/components/Header';
import Footer from '@/components/Footer';
import InvoicePayment from '@/components/InvoicePayment';
import { Card, CardContent } from '@/components/ui/card';
import { Button } from '@/components/ui/button';
import { AlertCircle, RefreshCw } from 'lucide-react';

// Error Boundary to catch any uncaught errors
interface ErrorBoundaryState {
  hasError: boolean;
  error: Error | null;
}

class PaymentErrorBoundary extends React.Component<
  { children: React.ReactNode },
  ErrorBoundaryState
> {
  constructor(props: { children: React.ReactNode }) {
    super(props);
    this.state = { hasError: false, error: null };
  }

  static getDerivedStateFromError(error: Error): ErrorBoundaryState {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: React.ErrorInfo) {
    console.error('[PaymentPage] Error caught by boundary:', error);
    console.error('[PaymentPage] Error info:', errorInfo);
  }

  handleRetry = () => {
    this.setState({ hasError: false, error: null });
    window.location.reload();
  };

  render() {
    if (this.state.hasError) {
      return (
        <Card className="max-w-md mx-auto">
          <CardContent className="py-12 text-center">
            <AlertCircle className="h-10 w-10 text-destructive mx-auto mb-4" />
            <h3 className="text-lg font-semibold text-foreground mb-2">Something went wrong</h3>
            <p className="text-sm text-muted-foreground mb-2">
              We encountered an error loading this page.
            </p>
            {this.state.error && (
              <p className="text-xs text-muted-foreground mb-4 font-mono p-2 bg-muted rounded">
                {this.state.error.message}
              </p>
            )}
            <Button onClick={this.handleRetry} variant="outline">
              <RefreshCw className="mr-2 h-4 w-4" />
              Try Again
            </Button>
          </CardContent>
        </Card>
      );
    }

    return this.props.children;
  }
}

const PaymentPage: React.FC = () => {
  return (
    <div className="min-h-screen flex flex-col bg-background">
      <Header />
      <main className="flex-1 container py-10">
        <PaymentErrorBoundary>
          <InvoicePayment />
        </PaymentErrorBoundary>
      </main>
      <Footer />
    </div>
  );
};

    // PAY INVOICE â€” USDC
    // Tiered fee: flat $2 below $200 USDC, 1% above
    // Fee is always paid by the client, never the creator
    // ----------------------------------------------------------
    pub fn pay_invoice(
        ctx: Context<PayInvoice>,
        invoice_id: String,
        payment_amount: u64,
    ) -> Result<()> {
        let invoice = &ctx.accounts.invoice;
        require!(payment_amount > 0, ErrorCode::InvalidPaymentAmount);
        require!(invoice.invoice_id == invoice_id, ErrorCode::InvalidInvoiceId);
        require!(
            invoice.status == InvoiceStatus::Unpaid || invoice.status == InvoiceStatus::PartiallyPaid,
            ErrorCode::InvoiceNotPayable
        );
        require!(invoice.token_type == TokenType::USDC, ErrorCode::TokenTypeMismatch);
        let remaining = invoice.amount.checked_sub(invoice.amount_paid).ok_or(ErrorCode::MathOverflow)?;
        require!(payment_amount <= remaining, ErrorCode::PaymentExceedsRemaining);

        // Tiered fee for USDC â€” threshold is exact in micro-units
        let fee = if payment_amount < TIER_THRESHOLD_USDC {
            FLAT_FEE_USDC // $2 flat
        } else {
            payment_amount
                .checked_mul(PERCENTAGE_FEE_BPS)
                .ok_or(ErrorCode::MathOverflow)?
                .checked_div(10000)
                .ok_or(ErrorCode::MathOverflow)?
        };

        // Transfer payment_amount to escrow
        token::transfer(
            CpiContext::new(ctx.accounts.token_program.to_account_info(), Transfer {
                from: ctx.accounts.client_token.to_account_info(),
                to: ctx.accounts.escrow.to_account_info(),
                authority: ctx.accounts.client.to_account_info(),
            }),
            payment_amount,
        )?;

        // Transfer fee to platform treasury
        if fee > 0 {
            token::transfer(
                CpiContext::new(ctx.accounts.token_program.to_account_info(), Transfer {
                    from: ctx.accounts.client_token.to_account_info(),
                    to: ctx.accounts.platform_treasury_token.to_account_info(),
                    authority: ctx.accounts.client.to_account_info(),
                }),
                fee,
            )?;
        }

        let invoice = &mut ctx.accounts.invoice;
        let new_amount_paid = invoice.amount_paid.checked_add(payment_amount).ok_or(ErrorCode::MathOverflow)?;
        invoice.amount_paid = new_amount_paid;
        if invoice.client.is_none() { invoice.client = Some(ctx.accounts.client.key()); }
        if new_amount_paid == invoice.amount {
            invoice.status = InvoiceStatus::FullyPaid;
            invoice.paid_at = Some(Clock::get()?.unix_timestamp);
        } else {
            invoice.status = InvoiceStatus::PartiallyPaid;
        }
        Ok(())
    }

    // RELEASE FUNDS â€” USDC
    // Requires either:
    //   (a) client approved, OR
    //   (b) 7-day timeout has passed since request
    // Enforces installment schedule if set
    // ----------------------------------------------------------
    pub fn release_funds(
        ctx: Context<ReleaseFunds>,
        invoice_id: String,
        release_amount: u64,
    ) -> Result<()> {
        let invoice = &ctx.accounts.invoice;
        require!(release_amount > 0, ErrorCode::InvalidReleaseAmount);
        require!(invoice.creator == ctx.accounts.creator.key(), ErrorCode::Unauthorized);
        require!(invoice.invoice_id == invoice_id, ErrorCode::InvalidInvoiceId);
        require!(
            invoice.status == InvoiceStatus::PartiallyPaid || invoice.status == InvoiceStatus::FullyPaid,
            ErrorCode::InvoiceNotReleasable
        );
        require!(invoice.token_type == TokenType::USDC, ErrorCode::TokenTypeMismatch);
        require!(invoice.status != InvoiceStatus::Disputed, ErrorCode::InvoiceDisputed);

        // Dual approval check: must have request + either approval or timeout
        require!(invoice.completion_requested_at.is_some(), ErrorCode::NoReleaseRequested);
        let requested_at = invoice.completion_requested_at.unwrap();
        let now = Clock::get()?.unix_timestamp;
        let timeout_passed = now - requested_at > AUTO_RELEASE_TIMEOUT;
        require!(
            invoice.client_approved || timeout_passed,
            ErrorCode::AwaitingClientApproval
        );

        // Enforce installment schedule if set
        if !invoice.release_schedule.is_empty() {
            let idx = invoice.current_release_index as usize;
            require!(idx < invoice.release_schedule.len(), ErrorCode::AllInstallmentsReleased);
            let installment_bps = invoice.release_schedule[idx] as u64;
            let installment_amount = invoice.amount_paid
                .checked_mul(installment_bps).ok_or(ErrorCode::MathOverflow)?
                .checked_div(10000).ok_or(ErrorCode::MathOverflow)?;
            require!(release_amount == installment_amount, ErrorCode::InvalidInstallmentAmount);
        }

        let releasable = invoice.amount_paid.checked_sub(invoice.amount_released).ok_or(ErrorCode::MathOverflow)?;
        require!(release_amount <= releasable, ErrorCode::ReleaseExceedsAvailable);

        let creator_key = ctx.accounts.creator.key();
        let invoice_id_bytes = invoice.invoice_id.as_bytes();
        let escrow_bump = invoice.escrow_bump;
        let bump_arr = [escrow_bump];
        let seeds = &[b"escrow", invoice_id_bytes, creator_key.as_ref(), &bump_arr];
        let signer_seeds: &[&[&[u8]]] = &[seeds];

        token::transfer(
            CpiContext::new_with_signer(ctx.accounts.token_program.to_account_info(), Transfer {
                from: ctx.accounts.escrow.to_account_info(),
                to: ctx.accounts.creator_token.to_account_info(),
                authority: ctx.accounts.escrow.to_account_info(),
            }, signer_seeds),
            release_amount,
        )?;

        let invoice = &mut ctx.accounts.invoice;
        let new_amount_released = invoice.amount_released.checked_add(release_amount).ok_or(ErrorCode::MathOverflow)?;
        invoice.amount_released = new_amount_released;

        if !invoice.release_schedule.is_empty() {
            invoice.current_release_index = invoice.current_release_index.checked_add(1).ok_or(ErrorCode::MathOverflow)?;
        }

        // Reset approval state for next installment
        invoice.completion_requested_at = None;
        invoice.client_approved = false;

        if new_amount_released == invoice.amount_paid && invoice.amount_paid == invoice.amount {
            invoice.status = InvoiceStatus::Released;
            invoice.released_at = Some(Clock::get()?.unix_timestamp);

            // Close escrow account and return rent to creator
            let creator_key = ctx.accounts.creator.key();
            let invoice_id_bytes = ctx.accounts.invoice.invoice_id.as_bytes();
            let escrow_bump = ctx.accounts.invoice.escrow_bump;
            let bump_arr = [escrow_bump];
            let seeds = &[b"escrow", invoice_id_bytes, creator_key.as_ref(), &bump_arr];
            let signer_seeds: &[&[&[u8]]] = &[seeds];
            token::close_account(
                CpiContext::new_with_signer(ctx.accounts.token_program.to_account_info(), CloseAccount {
                    account: ctx.accounts.escrow.to_account_info(),
                    destination: ctx.accounts.creator.to_account_info(),
                    authority: ctx.accounts.escrow.to_account_info(),
                }, signer_seeds),
            )?;
        }
        Ok(())
    }

        let releasable = invoice.amount_paid.checked_sub(invoice.amount_released).ok_or(ErrorCode::MathOverflow)?;
        require!(release_amount <= releasable, ErrorCode::ReleaseExceedsAvailable);
        let new_amount_released = invoice.amount_released.checked_add(release_amount).ok_or(ErrorCode::MathOverflow)?;
        let is_final_release = new_amount_released == invoice.amount_paid && invoice.amount_paid == invoice.amount;

        let invoice_info = ctx.accounts.invoice.to_account_info();
        let creator_info = ctx.accounts.creator.to_account_info();
        let rent = Rent::get()?;
        let rent_exempt_minimum = rent.minimum_balance(8 + Invoice::LEN);
        let invoice_lamports = invoice_info.lamports();
        let available_lamports = invoice_lamports.checked_sub(rent_exempt_minimum).ok_or(ErrorCode::MathOverflow)?;

        if is_final_release {
            **invoice_info.try_borrow_mut_lamports()? = rent_exempt_minimum;
            **creator_info.try_borrow_mut_lamports()? = creator_info.lamports()
                .checked_add(available_lamports).ok_or(ErrorCode::MathOverflow)?;
        } else {
            require!(release_amount <= available_lamports, ErrorCode::ReleaseExceedsAvailable);
            **invoice_info.try_borrow_mut_lamports()? = invoice_lamports
                .checked_sub(release_amount).ok_or(ErrorCode::MathOverflow)?;
            **creator_info.try_borrow_mut_lamports()? = creator_info.lamports()
                .checked_add(release_amount).ok_or(ErrorCode::MathOverflow)?;
        }

        let invoice = &mut ctx.accounts.invoice;
        invoice.amount_released = new_amount_released;
        if !invoice.release_schedule.is_empty() {
            invoice.current_release_index = invoice.current_release_index.checked_add(1).ok_or(ErrorCode::MathOverflow)?;
        }
        invoice.completion_requested_at = None;
        invoice.client_approved = false;

        if is_final_release {
            invoice.status = InvoiceStatus::Released;
            invoice.released_at = Some(Clock::get()?.unix_timestamp);
        }
        Ok(())
    }

    // ----------------------------------------------------------
    // RESOLVE DISPUTE (KENNY ONLY)
    // Kenny splits funds manually after reviewing evidence
    // creator_bps + client_bps must equal 10000
    // Only remaining unreleased installments are split
    // ----------------------------------------------------------
    pub fn resolve_dispute(
        ctx: Context<ResolveDispute>,
        invoice_id: String,
        creator_bps: u16,
        client_bps: u16,
    ) -> Result<()> {
        let invoice = &ctx.accounts.invoice;
        require!(invoice.invoice_id == invoice_id, ErrorCode::InvalidInvoiceId);
        require!(invoice.status == InvoiceStatus::Disputed, ErrorCode::NotDisputed);
        require!(ctx.accounts.kenny.key() == PLATFORM_TREASURY, ErrorCode::Unauthorized);
        // Validate client token account belongs to the client who paid
        if let Some(client_key) = invoice.client {
            require!(ctx.accounts.client_token.owner == client_key, ErrorCode::InvalidOwner);
        }
        require!(creator_bps as u32 + client_bps as u32 == 10000, ErrorCode::InvalidSplit);
        require!(invoice.token_type == TokenType::USDC, ErrorCode::TokenTypeMismatch);

        // Funds available for split = amount_paid - amount_released
        let splittable = invoice.amount_paid
            .checked_sub(invoice.amount_released)
            .ok_or(ErrorCode::MathOverflow)?;
        require!(splittable > 0, ErrorCode::NoFundsToSplit);

        let creator_share = splittable
            .checked_mul(creator_bps as u64).ok_or(ErrorCode::MathOverflow)?
            .checked_div(10000).ok_or(ErrorCode::MathOverflow)?;
        let client_share = splittable
            .checked_sub(creator_share).ok_or(ErrorCode::MathOverflow)?; // Remainder goes to client

        let creator_key = ctx.accounts.creator.key();
        let invoice_id_bytes = invoice.invoice_id.as_bytes();
        let escrow_bump = invoice.escrow_bump;
        let bump_arr = [escrow_bump];
        let seeds = &[b"escrow", invoice_id_bytes, creator_key.as_ref(), &bump_arr];
        let signer_seeds: &[&[&[u8]]] = &[seeds];

        // Transfer creator's share
        if creator_share > 0 {
            token::transfer(
                CpiContext::new_with_signer(ctx.accounts.token_program.to_account_info(), Transfer {
                    from: ctx.accounts.escrow.to_account_info(),
                    to: ctx.accounts.creator_token.to_account_info(),
                    authority: ctx.accounts.escrow.to_account_info(),
                }, signer_seeds),
                creator_share,
            )?;
        }

        // Transfer client's share
        if client_share > 0 {
            token::transfer(
                CpiContext::new_with_signer(ctx.accounts.token_program.to_account_info(), Transfer {
                    from: ctx.accounts.escrow.to_account_info(),
                    to: ctx.accounts.client_token.to_account_info(),
                    authority: ctx.accounts.escrow.to_account_info(),
                }, signer_seeds),
                client_share,
            )?;
        }

        let invoice = &mut ctx.accounts.invoice;
        invoice.amount_released = invoice.amount_released.checked_add(creator_share).ok_or(ErrorCode::MathOverflow)?;
        invoice.status = InvoiceStatus::Released;
        invoice.released_at = Some(Clock::get()?.unix_timestamp);

        // Close escrow
        let creator_key = ctx.accounts.creator.key();
        let invoice_id_bytes = ctx.accounts.invoice.invoice_id.as_bytes();
        let escrow_bump = ctx.accounts.invoice.escrow_bump;
        let bump_arr = [escrow_bump];
        let seeds = &[b"escrow", invoice_id_bytes, creator_key.as_ref(), &bump_arr];
        let signer_seeds: &[&[&[u8]]] = &[seeds];
        token::close_account(
            CpiContext::new_with_signer(ctx.accounts.token_program.to_account_info(), CloseAccount {
                account: ctx.accounts.escrow.to_account_info(),
                destination: ctx.accounts.creator.to_account_info(),
                authority: ctx.accounts.escrow.to_account_info(),
            }, signer_seeds),
        )?;

        Ok(())
    }
}

#[account]
pub struct Invoice {
    pub bump: u8,
    pub escrow_bump: u8,
    pub invoice_id: String,               // max 32
    pub creator: Pubkey,
    pub client: Option<Pubkey>,
    pub amount: u64,
    pub amount_paid: u64,
    pub amount_released: u64,
    pub token_type: TokenType,
    pub description: String,              // max 256
    pub client_email: String,             // max 128
    pub status: InvoiceStatus,
    pub created_at: i64,
    pub paid_at: Option<i64>,
    pub released_at: Option<i64>,
    // Release schedule
    pub release_schedule: Vec<u16>,       // basis points, sums to 10000
    pub current_release_index: u8,
    // Dual approval
    pub completion_requested_at: Option<i64>,
    pub client_approved: bool,
    // Dispute
    pub dispute_raised_at: Option<i64>,
}

impl Invoice {
    pub const LEN: usize =
        1 +           // bump
        1 +           // escrow_bump
        (4 + 32) +    // invoice_id String
        32 +          // creator Pubkey
        (1 + 32) +    // client Option<Pubkey>
        8 +           // amount
        8 +           // amount_paid
        8 +           // amount_released
        1 +           // token_type
        (4 + 256) +   // description String
        (4 + 128) +   // client_email String
        1 +           // status
        8 +           // created_at
        (1 + 8) +     // paid_at Option<i64>
        (1 + 8) +     // released_at Option<i64>
        (4 + MAX_INSTALLMENTS * 2) + // release_schedule Vec<u16>
        1 +           // current_release_index
        (1 + 8) +     // completion_requested_at Option<i64>
        1 +           // client_approved bool
        (1 + 8);      // dispute_raised_at Option<i64>
}
